# ---------- 5. Форматтеры ----------
def money(amount):
    return str(amount) + " ₽"

def job_text(j):
    verified = " ✅" if j.get("verified") else ""
    reqs = " · ".join(j.get("requirements", ["Оплата сразу после смены"]))
    return (
        j.get("category_emoji", "📋") + " <b>" + j.get("title", "") + "</b>" + verified + "\n"
        "💰 <b>" + money(j.get("pay", 0)) + "</b> за смену\n"
        "📍 " + j.get("metro", "") + " · " + j.get("date", "") + " · " + j.get("time", "") + "\n"
        "👥 Мест: " + j.get("slots", "") + "\n"
        "🏢 " + j.get("employer", "") + "\n"
        "\n"
        "📝 " + j.get("description", "") + "\n"
        "\n"
        "📋 Требования: " + reqs + "\n"
    )

# ---------- 7. Сессии ----------
sessions = {}

def session(chat_id):
    if chat_id not in sessions:
        sessions[chat_id] = {"role": None, "view": "start"}
    return sessions[chat_id]

# ---------- 8. Хендлеры команд и текста ----------
@bot.message_handler(commands=["start", "menu"])
def cmd_start(message):
    chat_id = message.chat.id
    s = session(chat_id)
    s["view"] = "start"
    bot.send_message(
        chat_id,
        "Привет! Это демо-бот Подработка.МСК.\n\n"
        "Здесь всё работает в режиме имитации: смены, отклики и статусы — "
        "демонстрационные данные без реальной базы.\n\n"
        "Выберите, кем вы заходите:",
        reply_markup=chat_keyboard(),
    )

@bot.message_handler(func=lambda m: m.text == "Я ищу подработку (исполнитель)")

def main_keyboard(role):
    markup = ReplyKeyboardMarkup(resize_keyboard=True)
    if role == "employer":
        markup.row(KeyboardButton("Мои смены"), KeyboardButton("Кандидаты"))
        markup.row(KeyboardButton("Чаты"), KeyboardButton("Профиль"))
        markup.row(KeyboardButton("Тарифы"), KeyboardButton("Помощь"))
    elif role == "worker":
        markup.row(KeyboardButton("Найти смену"), KeyboardButton("Сегодня"), KeyboardButton("Завтра"))
        markup.row(KeyboardButton("Избранное"), KeyboardButton("Мои отклики"))
        markup.row(KeyboardButton("Чаты"), KeyboardButton("Профиль"))
        markup.row(KeyboardButton("Тарифы"), KeyboardButton("Помощь"))
    else:
        markup.row(KeyboardButton("Привязать аккаунт"), KeyboardButton("Регистрация"))
        markup.row(KeyboardButton("Найти смену"), KeyboardButton("Тарифы"), KeyboardButton("Помощь"))
    return markup

def handle_text(message):

    text = (message.text or "").strip()
    s = session(chat_id)

    if s["view"] == "start" or s.get("role") is None:
        bot.send_message(
            chat_id,
            "Кем вы будете на платформе?",
            reply_markup=chat_keyboard(),
        )
        return

    role = s.get("role")
    bot.send_message(
        chat_id,
        "Выберите кнопку в меню ниже или /start — перезапустить демо.",
        reply_markup=main_keyboard(role),
    )
